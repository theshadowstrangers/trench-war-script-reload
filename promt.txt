-- Очистка подсветки
local function clearHighlights()
    for _, box in pairs(createdBoxes) do
        if box then box:Destroy() end
    end
    createdBoxes = {}
end

-- Закрытие
CloseButton.MouseButton1Click:Connect(function()
    isEspActive = false

-- Подсветка BoxHandleAdornment
local function applyBoxEsp(part, color)
    if not part:IsA("BasePart") then return end
    
    local box = Instance.new("BoxHandleAdornment")
    box.Size = part.Size
    box.AlwaysOnTop = true
    box.ZIndex = 10
    box.Color3 = color
    box.Transparency = 0.4
    box.Adornee = part
    box.Parent = game:GetService("CoreGui")
    
    table.insert(createdBoxes, box)
end

-- Обновление ESP

-- Логика клика по кнопке Touch-roto one
TouchButton.MouseButton1Click:Connect(function()
    -- Проверяем, есть ли у инжектора функция firetouchinterest
    if not firetouchinterest then
        TouchButton.Text = "No executor support!"
        task.wait(1.5)
        TouchButton.Text = "Touch-roto one"
        return
    end

    -- Находим ногу или торс игрока, чтобы "коснуться" объекта
    local player = game:GetService("Players").LocalPlayer
    local character = player.Character
    local rootPart = character and character:FindFirstChild("HumanoidRootPart")

    if not rootPart then
        TouchButton.Text = "No Character Found!"
        task.wait(1.5)
        TouchButton.Text = "Touch-roto one"
        return
    end

    -- Визуальный эффект нажатия (кнопка моргает синим во время работы)
    TouchButton.BackgroundColor3 = Color3.fromRGB(0, 120, 255)
    TouchButton.Text = "Touching..."

    local count = 0
    -- Пробегаемся по всем объектам в Workspace
    for _, obj in pairs(workspace:GetDescendants()) do
        -- Ищем именно блоки с именем Vidrio roto
        if obj.Name == "Vidrio roto" and obj:IsA("BasePart") then
            -- Проверяем, есть ли внутри TouchInterest (он же TouchTransmitter в API)
            if obj:FindFirstChildOfClass("TouchTransmitter") then
                -- Имитируем ОДИНАКОВЫЙ полноценный клик: коснулся (0) и убрал ногу (1)
                firetouchinterest(rootPart, obj, 0)
                task.wait() -- Микропауза, чтобы игра успела обработать касание
                firetouchinterest(rootPart, obj, 1)
                count = count + 1
            end
        end
    end

    -- Возвращаем кнопку в исходный вид и пишем сколько стекол активировали
    TouchButton.BackgroundColor3 = Color3.fromRGB(50, 200, 50)
    TouchButton.Text = "Done! ("..count..")"
    task.wait(2)
    TouchButton.BackgroundColor3 = Color3.fromRGB(70, 70, 90)
    TouchButton.Text = "Touch-roto one"
end)
